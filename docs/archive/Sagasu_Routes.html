config/routes.rb

# config/routes.rb
Rails.application.routes.draw do
  devise_for :users
  root to: "pages#home"                                        # 1, 2

  resources :projects, only: [:index, :new, :create, :show] do # 3
    resources :items, only: [:create]                          # 4
    resources :item_groups, only: [] do
      collection { post :generate }                            # 5  (AI)
    end
    resources :approvals, only: [:create]                      # 12
  end

  resources :items, only: [:update, :destroy]                  # 4, 6

  resources :item_groups, only: [:update] do                   # 6
    resources :group_suppliers, only: [:create] do             # 9
      collection { post :suggest }                             # 7  (AI)
    end
  end

  resources :group_suppliers, only: [:update]                  # 8

  resources :suppliers, only: [:index, :show] do               # 9, 10
    resources :reviews, only: [:create]                        # 11
  end

  resources :approvals, only: [:index, :update]                # 13, 14
end

Notes

  • Story 8: "remove" a supplier = set selected: false (keeps the AI suggestion for traceability) rather than deleting the row.
  • Story 12: creating an approval also sets projects.status to pending_approval.
  • Story 14: the approver sets status to approved or returned with a comment; the project status follows.
  • Access: only users with role: approver can reach approvals#index and approvals#update (Pundit policy).
  • Later: invitations and supplier proposals will add their own routes after the MVP.
